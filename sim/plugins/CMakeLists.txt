cmake_minimum_required(VERSION 3.10)
project(yaw_bridge)
find_package(gazebo REQUIRED)
set(CMAKE_CXX_STANDARD 17)
include_directories(${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
list(APPEND CMAKE_CXX_FLAGS "${GAZEBO_CXX_FLAGS}")
add_library(yaw_bridge SHARED yaw_bridge.cc)
target_link_libraries(yaw_bridge ${GAZEBO_LIBRARIES})
add_library(arms_bridge SHARED arms_bridge.cc)
target_link_libraries(arms_bridge ${GAZEBO_LIBRARIES})
add_library(track_bridge SHARED track_bridge.cc)
target_link_libraries(track_bridge ${GAZEBO_LIBRARIES})
